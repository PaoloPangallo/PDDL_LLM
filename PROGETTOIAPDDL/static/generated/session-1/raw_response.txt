===DOMAIN===
(define (domain ElvenRealm)
  (:requirements :strips :typing)
  (:types entity location_type route_type challenge_type item_type traveler_type)
  (:predicates
    (at ?x - traveler_type ?l - location_type)
    (can_access ?r - route_type)
    (has_challenge ?l - challenge_type)
    (contains ?l ?i - item_type)
    (requires ?c - challenge_type ?i - item_type))
  (:action ClimbSunlitCliffs
    :parameters (?traveler - traveler_type ?route - route_type)
    :precondition (and (at ?traveler Port of Murias))
    :effect (and (at ?traveler Sunlit Cliffs)))
  (:action CrossShadowedMarsh
    :parameters (?traveler - traveler_type ?route - route_type)
    :precondition (and (at ?traveler Port of Murias))
    :effect (and (at ?traveler Shadowed Marsh)))
  (:action VisitTemple
    :parameters (?traveler - traveler_type ?location - location_type)
    :precondition (and (at ?traveler Sunlit Cliffs))
    :effect (and (at ?traveler Temple of Tides)))
  (:action RetrieveEmerald
    :parameters (?traveler - traveler_type ?item - item_type)
    :precondition (and (at ?traveler Temple of Tides))
    :effect (and (contains Temple of Tides Emerald of Dawn)))
)
===PROBLEM===
(define (problem QuestForEmerald)
  (:domain ElvenRealm)
  (:objects
    Port of Murias - location_type
    Sunlit Cliffs - route_type
    Shadowed Marsh - route_type
    Temple of Tides - location_type
    Sanctuary of Light - location_type)
  (:init
    (at Arin Port of Murias)
    (can_access Sunlit Cliffs)
    (can_access Shadowed Marsh)
    (has_challenge Temple of Tides))
  (:goal (and (at Arin Sanctuary of Light) (contains Temple of Tides Emerald of Dawn)))
)
